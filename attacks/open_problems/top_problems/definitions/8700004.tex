\section{Conjecture 1.5}\label{8700004}
{\small\sffamily UnsolvedMath ID 8700004 · Graph Theory}
\subsection{Definitions and mathematical statement}

\paragraph{Shared notation.}$\mathbb N=\{1,2,\ldots\}$, and $\mathbb Z,\mathbb Q,\mathbb R,\mathbb C$ denote the integers, rationals, reals and complex numbers. Any other conventions are specified locally.


Let $k\ge2$ and let $\alpha_1,\ldots,\alpha_k\ne0$ lie in a characteristic-zero field $K$, with $\alpha_i/\alpha_j$ not a root of unity for $i\ne j$. For $x_1,\ldots,x_k\in\mathbb Z$, put $M(x)=(\alpha_j^{x_i})_{1\le i,j\le k}$; negative exponents use inverses in $K$.
\paragraph{Statement recorded in the supplied source.}
Are there only finitely many $(x_2,\ldots,x_k)\in\mathbb Z^{k-1}$ such that $\det M(0,x_2,\ldots,x_k)=0$, every matrix formed by deleting one row has rank $k-1$, and every matrix formed by deleting one column has rank $k-1$? \sref{8700004}{2}
\subsection{Short English statement}
Does the nondegenerate exponential determinant equation with first exponent zero have only finitely many integer solutions?
